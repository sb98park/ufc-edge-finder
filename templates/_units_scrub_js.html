{# Shared Units-curve scrub behaviour. Included by both the app's Record tab and the landing page's record card, which now render the same chart from the same builder. Self-contained: it depends on nothing else on the page. Call initUnitsScrub(cfg) once per chart. Everything optional is optional -- the landing page has no dollar figure and no headline to snap back to, and passing nothing for them is the supported path, not a degraded one. #} // ===== Stable-DOM odometer ===== // Returns a set(text) that only moves the digit positions whose digit // actually changed. See _units_scrub_css.html for why the DOM must never // be rebuilt mid-gesture. function makeOdometer(el) { let shape = '', cells = []; function build(str) { el.textContent = ''; cells = []; for (const ch of str) { if (ch >= '0' && ch <= '9') { const cell = document.createElement('span'); cell.className = 'od-cell'; const col = document.createElement('span'); col.className = 'od-col'; for (let d = 0; d < 10; d++) { const s = document.createElement('span'); s.className = 'od-d'; s.textContent = d; col.appendChild(s); } // The cell holds all ten digits at once and only one is visible. // Left readable, a screen reader would announce "0123456789" per // position; the element's aria-label below carries the real value. cell.setAttribute('aria-hidden', 'true'); cell.appendChild(col); // Deliberately not '0': an empty marker guarantees the first set() // writes every position, so a column can never start out showing a // digit it was never told to show. cell._d = ''; el.appendChild(cell); cells.push(cell); } else { const lit = document.createElement('span'); lit.className = 'od-lit'; lit.textContent = ch; el.appendChild(lit); cells.push(null); } } } return function set(str) { // The accessible value, always the plain number. el.setAttribute('aria-label', str + 'U'); // Rebuild only when the SHAPE changes -- "+9.99" to "+10.02" gains a // column and has to be rebuilt, "+9.99" to "+8.12" does not. const sig = str.replace(/[0-9]/g, '#'); // The second condition is the guard against anything else on the page // having written textContent over these cells since the last set(). // The landing page's reveal count-up does exactly that, and without // this the odometer would go on transforming nodes that are no longer // in the document -- silently, and only on that one page. if (sig !== shape || el.childNodes.length !== cells.length) { shape = sig; build(str); } let i = 0; for (const ch of str) { const cell = cells[i++]; if (!cell || cell._d === ch) continue; cell._d = ch; // Percentage of the COLUMN's own height: ten digits stacked means // one digit is 10%. No pixel measurement, so it survives a font // size that differs between the app and the landing page. cell.firstChild.style.transform = 'translateY(-' + (ch * 10) + '%)'; } }; } // ===== Wiring ===== // cfg: { wrap, figure, points, dollars, total, readout } // wrap - element wrapping the svg (gets .scrubbing) // figure - element holding the cumulative number // points -
